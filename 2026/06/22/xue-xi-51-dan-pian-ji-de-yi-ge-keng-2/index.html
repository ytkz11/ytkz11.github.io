学习51单片机的一个坑2 | 普中51 + STC89C52 数码管“0~9”逐字测试程序

2026年6月22日嵌入式
/***********************************************************
  普中51 + STC89C52  数码管“0~9”逐字测试程序
  功能：只点亮第1位数码管，循环显示 0~9
  用途：精准判断是段码错误、位选错误，还是硬件段损坏
***********************************************************/

#include <reg52.h>
#include <intrins.h>

sbit DU = P2^6;   // 段选锁存
sbit WE = P2^7;   // 位选锁存

// ---------- 共阴数码管 0~9 段码表（高电平点亮）----------
unsigned char code seg_table[] = {
    0x3F,  // 0
    0x06,  // 1
    0x5B,  // 2
    0x4F,  // 3
    0x66,  // 4
    0x6D,  // 5
    0x7D,  // 6
    0x07,  // 7
    0x7F,  // 8
    0x6F   // 9
};
// 如果你的板子是共阳数码管，请把上面每个数取反（即 0xC0, 0xF9...）
// 但普中绝大多数是共阴，先按这个试。

// ========= 位选码（二选一，只留一个生效） =========
// 【方案A】74HC138译码器模式（优先试这个）
unsigned char code wei_code[] = {0x00, 0x01, 0x02, 0x03, 0x04, 0x05, 0x06, 0x07};

// 【方案B】直接位选模式（如果方案A全乱，就换这个）
// unsigned char code wei_code[] = {0xFE, 0xFD, 0xFB, 0xF7, 0xEF, 0xDF, 0xBF, 0x7F};
// ==================================================

// 延时函数（12MHz下约1ms）
void delay_ms(unsigned int xms) {
    unsigned int i, j;
    for (i = xms; i > 0; i--)
        for (j = 114; j > 0; j--);
}

void main() {
    unsigned char i;
    
    // 第一步：先固定选中第1位数码管（位选只送一次，后面只改段码）
    P0 = wei_code[0];   // 选中第1位（译码器模式送0x00，直接模式送0xFE）
    WE = 1;             // 打开位锁存
    _nop_(); _nop_();
    WE = 0;             // 锁存位选（之后第1位会一直保持选中）

    while (1) {
        for (i = 0; i < 10; i++) {   // i=0~9
            // 送段码（显示数字 i）
            P0 = seg_table[i];
            DU = 1;          // 打开段锁存
            _nop_(); _nop_();
            DU = 0;          // 锁存段码

            delay_ms(1000);  // 保持显示1秒，方便观察
        }
    }
}
